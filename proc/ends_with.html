   pure logical function ends_with(s, suffix) result(ok)
      character(len=*), intent(in) :: s, suffix
      integer :: ns, n
      ns = len_trim(suffix)
      n = len_trim(s)
      if (ns <= 0) then
         ok = .true.
      else if (n < ns) then
         ok = .false.
      else
         ok = (s(n-ns+1:n) == suffix(1:ns))
      end if
   end function ends_with
