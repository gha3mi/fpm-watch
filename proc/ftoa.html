ftoa

public pure function ftoa(x, precision) result(s)

Format a real using two decimal places by default and a leading zero.

Optional precision is clamped to 0..16. Fixed output that overflows the 64-character buffer falls back to scientific notation. Nonfinite values are not rejected here; typed real logging handles them separately.

   pure function ftoa(x, precision) result(s)
      real, intent(in) :: x
      integer, intent(in), optional :: precision
      character(len=:), allocatable :: s
      character(len=64) :: buf
      character(len=16) :: format
      integer :: digits
      digits = 2
      if (present(precision)) digits = max(0, min(precision, 16))
      write(format, "(a,i0,a)") "(f0.", digits, ")"
      write(buf, format) x
      if (index(buf, "*") > 0) write(buf, "(es24.16e3)") x
      s = trim(adjustl(buf))
      if (s(1:1) == ".") s = "0" // s
      if (s(1:2) == "-.") s = "-0" // s(2:)
   end function ftoa
