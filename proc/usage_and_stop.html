   subroutine usage_and_stop(msg)
      character(len=*), intent(in) :: msg
      call log_event("error", "cli.error", msg)
      if (.not. structured_logging()) write(error_unit,"(a)") "Use fpm-watch --watch-help for usage."
      stop 2
   end subroutine usage_and_stop
