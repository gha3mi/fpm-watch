   pure function from_c_string(buffer) result(path)
      character(kind=c_char), intent(in) :: buffer(:)
      character(len=:), allocatable :: path
      integer :: i, n
      n = 0
      do i = 1, size(buffer)
         if (buffer(i) == c_null_char) exit
         n = n + 1
      end do
      path = repeat(" ", n)
      do i = 1, n
         path(i:i) = buffer(i)
      end do
   end function from_c_string
