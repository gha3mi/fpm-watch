   pure function lower_ascii(s) result(r)
      character(len=*), intent(in) :: s
      character(len=len(s)) :: r
      integer :: k, c, da
      da = iachar("a") - iachar("A")
      do k = 1, len(s)
         c = iachar(s(k:k))
         if (c >= iachar("A") .and. c <= iachar("Z")) then
            r(k:k) = achar(c + da)
         else
            r(k:k) = s(k:k)
         end if
      end do
   end function lower_ascii
