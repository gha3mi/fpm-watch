execute_command

public subroutine execute_command(cmd, exitstat, interrupted, silent, stderr)

Execute a command while preserving user interruption requests.

The C wrapper waits for the child without ignoring console interrupts, as some implementations of synchronous execute_command_line do. Ordinary command failures are returned without requesting watcher termination. POSIX children share the foreground process group for interactive input; temporary signal handlers forward requests to the immediate child. On Windows, console requests terminate the child's job and its descendants. Uninterrupted command exits preserve the lifetime of background children. Original signal handlers are restored when the command finishes. Windows executables receive quoted argv directly; command scripts use a shell fallback. Silent output uses inherited null handles on Windows and shell redirection on POSIX.

   subroutine execute_command(cmd, exitstat, interrupted, silent, stderr)
      character(len=*), intent(in) :: cmd
      integer, intent(out) :: exitstat
      logical, intent(out), optional :: interrupted
      logical, intent(in), optional :: silent
      logical, intent(in), optional :: stderr
      integer(c_int) :: status, was_interrupted, launch_error, silent_mode
      character(len=:), allocatable :: effective_cmd

      silent_mode = 0_c_int
      if (present(stderr)) then
         if (stderr) silent_mode = 2_c_int
      end if
      if (present(silent)) then
         if (silent) silent_mode = 1_c_int
      end if
      effective_cmd = trim(cmd)
      if (.not. is_windows()) then
         if (silent_mode == 1_c_int) effective_cmd = effective_cmd // " >/dev/null 2>&1"
         if (silent_mode == 2_c_int) effective_cmd = effective_cmd // " 1>&2"
      end if
      status = fpm_watch_execute(effective_cmd // c_null_char, silent_mode, was_interrupted, launch_error)
      stop_requested = stop_requested .or. was_interrupted /= 0_c_int
      if (present(interrupted)) interrupted = was_interrupted /= 0_c_int
      if (launch_error /= 0_c_int) then
         call log_event("error", "command.launch_failed", "Cannot execute command: " // trim(cmd), &
                      [log_int("os_error", int(launch_error)), log_string("command", trim(cmd))])
         exitstat = 127
      else
         exitstat = int(status)
      end if
   end subroutine execute_command
